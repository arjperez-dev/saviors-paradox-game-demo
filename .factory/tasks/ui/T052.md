# T052 — Give the screens one ink theme at the screen's size

## Description

The one theme every screen draws with (coding-standards §11.6, §11.7): `InkTheme`, a widget at the
root that reads the screen's height once through a `LayoutBuilder` and gives every widget under it
an `InkThemeData` — the UI scale as the `UiScale` port, hu → logical pixels, the kit's lettering
roles in Patrick Hand, and every colour and stroke width the kit draws with, each the 06-ui
sandbox's value in a palette colour. A frame without a resize keeps the same `InkThemeData`
instance, so nothing allocates and `updateShouldNotify`, comparing the scale, rebuilds nothing.
`InkLettering` is the closed list of the kit's lettering roles, each a size in hu and a colour.

## Preconditions

- T051's `ScreenUiScale` on `feat/ui`. On `main`: `Palette` (`ink`, `paper`, `paperLight`) and
  `Lettering.style` / `Lettering.family` ('Patrick Hand') in `core/rendering/presentation/`.

## Constraints

- The sheet's shadow is darkness, `rgb(14,11,9)` at .55, in place of the sandbox's black, which is
  not in palette.md (operator, 2026-10-16).
- The box fills are `ttBox`'s, .78 at rest and .95 under the mouse, for every screen (operator,
  2026-10-16).
- `InkTheme.of` depends on the scope (`dependOnInheritedWidgetOfExactType`), so a resize rebuilds
  what reads it.

```dart
// lib/core/ui/presentation/ink_lettering.dart — new
import 'dart:ui';

import 'package:saviors_paradox/core/rendering/presentation/palette.dart';

/// The kit's lettering roles: each a size in hu and a colour, as the 06-ui
/// sandbox letters them (`hudText`, `keyCap`, `pmFooter`, `ttMain`,
/// `ttConfirm`).
enum InkLettering {
  // pmFooter, ttMain: 30, ink.
  button(30, Palette.ink),
  // ttMain: a faint button's label, ink at .45.
  buttonFaint(30, Color.fromRGBO(26, 23, 21, 0.45)),
  // pmFooter, ttSlots, ttSettings: 24, ink at .7.
  hint(24, Color.fromRGBO(26, 23, 21, 0.7)),
  // keyCap: 20, ink.
  keyCap(20, Palette.ink),
  // ttConfirm: the question, 42, ink.
  question(42, Palette.ink),
  // ttConfirm: the note under it, 28, ink at .8.
  questionNote(28, Color.fromRGBO(26, 23, 21, 0.8));

  const InkLettering(this.size, this.colour);

  /// The size, in hu.
  final double size;

  final Color colour;
}
```

```dart
// lib/core/ui/presentation/ink_theme_data.dart — new
import 'package:flutter/painting.dart';
import 'package:saviors_paradox/core/contracts/ports/ui_scale.dart';
import 'package:saviors_paradox/core/rendering/presentation/lettering.dart';
import 'package:saviors_paradox/core/rendering/presentation/palette.dart';
import 'package:saviors_paradox/core/ui/presentation/ink_lettering.dart';
import 'package:saviors_paradox/core/ui/presentation/screen_ui_scale.dart';

/// The look of every screen at the screen's size: the UI scale, the
/// lettering, and the colours and stroke widths the kit draws with, in hu
/// (06-ui/ui.md §3; coding-standards.md §11.6). Colours are palette.md's;
/// the values are the 06-ui sandbox's.
final class InkThemeData {
  const InkThemeData(this._scale);

  final ScreenUiScale _scale;

  /// The UI frame at this screen's size, for hud and map to call.
  UiScale get scale => _scale;

  /// [hu] in logical pixels.
  double pixels(double hu) => hu * _scale.pixelsPerHu;

  /// The lettering of [role], scaled to this screen.
  TextStyle lettering(InkLettering role) =>
      Lettering.style(size: pixels(role.size), color: role.colour);

  /// The width of [text] in [role]'s lettering, in hu: a box sized to its
  /// text, as the sandbox's `measureText`.
  static double widthOf(String text, InkLettering role) {
    final painter = TextPainter(
      text: TextSpan(
        text: text,
        style: Lettering.style(size: role.size, color: role.colour),
      ),
      textDirection: TextDirection.ltr,
    )..layout();
    final width = painter.width;
    painter.dispose();
    return width;
  }

  // The paper sheet: the sandbox's `sheetAt`.
  static const double sheetBorderWidth = 3.2;
  static const double sheetInnerWidth = 1.2;
  static const Color sheetInnerColour = Color.fromRGBO(26, 23, 21, 0.4);
  // `shadowColor rgba(0,0,0,.55)`: darkness, the nearest palette colour.
  static const Color sheetShadowColour = Color.fromRGBO(14, 11, 9, 0.55);
  // `shadowBlur 34`: a canvas blur of 34 is a Gaussian of sigma 17.
  static const double sheetShadowSigma = 17;
  static const Color sheetLight = Color.fromRGBO(239, 230, 214, 0.32);
  static const Color sheetShade = Color.fromRGBO(14, 11, 9, 0.3);

  // A box's pencil frame: the sandbox's `hudBox`.
  static const double frameWidth = 1.7;
  static const Color frameColour = Color.fromRGBO(26, 23, 21, 0.62);
  static const double focusWidth = 3.6;
  static const Color focusColour = Palette.ink;
  static const double focusInnerWidth = 1;
  static const Color focusInnerColour = Color.fromRGBO(26, 23, 21, 0.35);
  static const double dashWidth = 1.4;
  static const Color dashColour = Color.fromRGBO(26, 23, 21, 0.45);
  static const double dashOn = 7;
  static const double dashOff = 6;

  // A box's paper: `ttBox`, `ttMain`'s faint buttons, `hudBox`'s dashed
  // box, `keyCap`, `ttConfirm`.
  static const Color paperFill = Color.fromRGBO(239, 230, 214, 0.78);
  static const Color paperFillHovered = Color.fromRGBO(239, 230, 214, 0.95);
  static const Color faintFill = Color.fromRGBO(212, 204, 190, 0.55);
  static const Color faintFillHovered = Color.fromRGBO(212, 204, 190, 0.75);
  static const Color dashedFill = Color.fromRGBO(239, 230, 214, 0.28);
  static const Color dashedFillHovered = Color.fromRGBO(26, 23, 21, 0.08);
  static const Color keyCapFill = Palette.paperLight;
  static const Color questionFill = Color.fromRGBO(239, 230, 214, 0.97);
  // pmConfirm: the darker wash over the sheet behind a question.
  static const Color questionWash = Color.fromRGBO(14, 11, 9, 0.45);

  // The hold ring: the cinematics sandbox's `skipOverlay`.
  static const double ringBaseWidth = 1.4;
  static const double ringArcWidth = 4.2;
  static const Color ringOnPaper = Palette.ink;
  static const Color ringBaseOnPaper = Color.fromRGBO(26, 23, 21, 0.3);
  static const Color ringOnDark = Color.fromRGBO(239, 230, 214, 0.9);
  static const Color ringBaseOnDark = Color.fromRGBO(239, 230, 214, 0.3);
}
```

```dart
// lib/core/ui/presentation/ink_theme.dart — new
import 'package:flutter/widgets.dart';
import 'package:saviors_paradox/core/ui/presentation/ink_theme_data.dart';
import 'package:saviors_paradox/core/ui/presentation/screen_ui_scale.dart';

/// Gives every widget under it the look of the screens at the screen's
/// size: the one place the kit reads the screen's height
/// (06-ui/ui.md §3; coding-standards.md §11.6, §11.7).
final class InkTheme extends StatefulWidget {
  const InkTheme({required this.child, super.key});

  final Widget child;

  /// The look of the nearest [InkTheme] above [context].
  static InkThemeData of(BuildContext context) {
    final scope = context.dependOnInheritedWidgetOfExactType<_InkThemeScope>();
    if (scope == null) {
      throw StateError('No InkTheme above this widget.');
    }
    return scope.data;
  }

  @override
  State<InkTheme> createState() => _InkThemeState();
}

final class _InkThemeState extends State<InkTheme> {
  InkThemeData? _data;
  double? _screenHeight;

  @override
  Widget build(BuildContext context) => LayoutBuilder(
    builder: (context, constraints) => _InkThemeScope(
      data: _dataFor(constraints.maxHeight),
      child: widget.child,
    ),
  );

  // A frame without a resize keeps the same data, so nothing allocates.
  InkThemeData _dataFor(double screenHeight) {
    final current = _data;
    if (current != null && _screenHeight == screenHeight) {
      return current;
    }
    _screenHeight = screenHeight;
    return _data = InkThemeData(ScreenUiScale(screenHeight: screenHeight));
  }
}

final class _InkThemeScope extends InheritedWidget {
  const _InkThemeScope({required this.data, required super.child});

  final InkThemeData data;

  @override
  bool updateShouldNotify(_InkThemeScope oldWidget) =>
      oldWidget.data.scale != data.scale;
}
```
