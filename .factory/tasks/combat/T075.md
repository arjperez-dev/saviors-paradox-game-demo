# T075 — Count down the status effects on every body

## Description

`StatusEffects`, the aggregate root that holds every status effect on every body of the zone
(glossary §4), with each effect's time left.

**Invariants it guards:**
- at most one entry per body and kind;
- every entry has time left;
- an effect with no time is refused;
- a new effect never shortens a longer one already running: it extends a shorter one, and
  otherwise is refused;
- entries expire in the order they were applied.

**Domain events it records:** `StatusStarted` (an effect began, or was extended to the duration
given), `StatusCleared` and `StatusExpired`. `apply` and `clear` return the event of their call;
`advance` adds its `StatusExpired` events to a list the caller owns, so the per-frame path
allocates only on expiry.

## Constraints

- **The aggregate boundary.** It has no entity type: its entries are three private parallel lists
  (`_targets`, `_kinds`, `_left`) that never leave it. Every change goes through `apply`, `clear`
  or `advance`; outside code reads only `has` and `remaining`.
- `advance` allocates nothing but the events of what expired: indexed loops, compaction in place.
- Domain events follow T073's `Constraints` (every field in `==`).

```dart
/// What happened to the status effects of a body.
sealed class StatusEffectEvent {
  const StatusEffectEvent();
}

/// [kind] began on [target], or was extended, to last [duration].
final class StatusStarted extends StatusEffectEvent {
  const StatusStarted({
    required this.target,
    required this.kind,
    required this.duration,
  });

  final EntityId target;
  final StatusEffectKind kind;
  final Seconds duration;

  @override
  bool operator ==(Object other) =>
      other is StatusStarted &&
      other.target == target &&
      other.kind == kind &&
      other.duration == duration;

  @override
  int get hashCode => Object.hash(target, kind, duration);
}

/// [kind] was cleared from [target] before its time ran out.
final class StatusCleared extends StatusEffectEvent {
  const StatusCleared({required this.target, required this.kind});

  final EntityId target;
  final StatusEffectKind kind;

  @override
  bool operator ==(Object other) =>
      other is StatusCleared && other.target == target && other.kind == kind;

  @override
  int get hashCode => Object.hash(target, kind);
}

/// [kind]'s time on [target] ran out.
final class StatusExpired extends StatusEffectEvent {
  const StatusExpired({required this.target, required this.kind});

  final EntityId target;
  final StatusEffectKind kind;

  @override
  bool operator ==(Object other) =>
      other is StatusExpired && other.target == target && other.kind == kind;

  @override
  int get hashCode => Object.hash(target, kind);
}
```

```dart
// lib/features/combat/domain/status_effects.dart — new
import 'package:saviors_paradox/core/contracts/kernel/entity_id.dart';
import 'package:saviors_paradox/core/contracts/kernel/seconds.dart';
import 'package:saviors_paradox/core/contracts/ports/status_effect_kind.dart';
import 'package:saviors_paradox/features/combat/domain/status_effect_event.dart';

/// Every status effect on every body, and the time each has left.
final class StatusEffects {
  // Parallel lists, one entry per body and kind, in the order applied.
  final List<EntityId> _targets = [];
  final List<StatusEffectKind> _kinds = [];
  final List<Seconds> _left = [];

  bool has(EntityId target, StatusEffectKind kind) =>
      _indexOf(target, kind) >= 0;

  Seconds remaining(EntityId target, StatusEffectKind kind) {
    final index = _indexOf(target, kind);
    return index < 0 ? Seconds.zero : _left[index];
  }

  /// Starts [kind] on [target] for [duration], or extends it to
  /// [duration]; null when [duration] is no time or does not outlast the
  /// effect already running.
  StatusStarted? apply(
    EntityId target,
    StatusEffectKind kind,
    Seconds duration,
  ) {
    if (duration <= Seconds.zero) {
      return null;
    }
    final index = _indexOf(target, kind);
    if (index >= 0 && duration <= _left[index]) {
      return null;
    }
    if (index < 0) {
      _targets.add(target);
      _kinds.add(kind);
      _left.add(duration);
    } else {
      _left[index] = duration;
    }
    return StatusStarted(target: target, kind: kind, duration: duration);
  }

  /// Ends [kind] on [target] at once; null when it was not running.
  StatusCleared? clear(EntityId target, StatusEffectKind kind) {
    final index = _indexOf(target, kind);
    if (index < 0) {
      return null;
    }
    _targets.removeAt(index);
    _kinds.removeAt(index);
    _left.removeAt(index);
    return StatusCleared(target: target, kind: kind);
  }

  /// Takes [dt] from every effect, adding one [StatusExpired] to [expired]
  /// for each whose time runs out, in the order they were applied.
  void advance(Seconds dt, List<StatusExpired> expired) {
    var kept = 0;
    for (var i = 0; i < _left.length; i++) {
      final left = _left[i] - dt;
      if (left <= Seconds.zero) {
        expired.add(StatusExpired(target: _targets[i], kind: _kinds[i]));
        continue;
      }
      _targets[kept] = _targets[i];
      _kinds[kept] = _kinds[i];
      _left[kept] = left;
      kept++;
    }
    _targets.length = kept;
    _kinds.length = kept;
    _left.length = kept;
  }

  int _indexOf(EntityId target, StatusEffectKind kind) {
    for (var i = 0; i < _targets.length; i++) {
      if (_targets[i] == target && _kinds[i] == kind) {
        return i;
      }
    }
    return -1;
  }
}
```
