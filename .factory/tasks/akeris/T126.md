# T126 — Carry Akeris along the walls on his chain

## Description

The wall half of the chain's hold on his body (`hook-blades.md` §3, `controls.md` G20). Chain says
only where the blade bit; how his body hangs from it is akeris's (planner wave-05
`chain-points-are-bites`).

- **`PulledToward(bite)`**: pulled across in 0.3 s, eased, toward `Vec2(bite.x, bite.y + 250)` (the
  sandbox's `wallGrabStep`). The wall stops him beside it. Then he hangs.
- **`ClimbingTo(bite)`**: a stretch in 0.5 s toward `Vec2(bite.x, bite.y + 200)`, 50 su a click (the
  sandbox's `wallClimb`; operator, 2026-10-17). Then he hangs.
  - **The mantle:** when the bite is the wall's top edge (stone 8 su into the wall just below the
    bite, none just above), he climbs onto the top instead, 40 su past the edge, in 0.62 s
    (animations.md §4). There he stands, and the chain is released.
- **Hanging** (`Hung(hold: wall)`, or after a pull or a stretch): he holds still. Space jumps him
  off, back and up (360, 820 su/s, the sandbox's `wallHang`).

He moves along the wall by `_Body.sweepBy`: up or down first, then across, so the wall stops him a
skin short. The mantle rises beside the wall, then crosses over its top. A hit never makes him let
go. A tether different from the one a posture followed hands him to the posture it names. Letting go
of a ring (T125) ends through the same `_release` helper as jumping off a wall.

**Domain event:** `ChainReleased()`, on jumping off and at the top of a mantle.

## Constraints

- As T125. No `as`: tethers are read with patterns.

```dart
// lib/features/akeris/domain/body.dart — `_Body` gains:

  /// Moves by [delta], up or down first and then across, stopping a skin
  /// short of what it meets: a wall stops him beside it.
  void sweepBy(Vec2 delta, Terrain terrain) {
    if (delta.y != 0) {
      final fraction = terrain.travel(box, Vec2(0, delta.y));
      feet = Vec2(feet.x, feet.y + _travelled(delta.y, fraction));
    }
    if (delta.x != 0) {
      final fraction = terrain.travel(box, Vec2(delta.x, 0));
      feet = Vec2(feet.x + _travelled(delta.x, fraction), feet.y);
    }
  }

  void faceToward(double x) =>
      facing = x < feet.x ? Facing.left : Facing.right;
```

```dart
// lib/features/akeris/domain/pulled_across.dart — new
part of 'akeris.dart';

// Pulled across a gap to the wall his blade bit: 0.3 s, eased, to hang
// 250 su below the bite (hook-blades.md §3; the sandbox's `wallGrabStep`).
final class _PulledAcross extends _Posture {
  Tether _following = const Slack();
  Vec2 _from = Vec2.zero;
  Seconds _elapsed = Seconds.zero;

  @override
  Stance stanceOf(Akeris akeris) => Stance.pulled;

  @override
  bool get isTethered => true;

  @override
  AkerisEvent? enter(Akeris akeris) {
    final body = akeris._body;
    _following = akeris._tether;
    _from = body.feet;
    _elapsed = Seconds.zero;
    body
      ..velocity = Vec2.zero
      ..isGrounded = false;
    if (_following case PulledToward(:final bite)) {
      body.faceToward(bite.x);
    }
    return null;
  }

  @override
  _Posture step(
    Akeris akeris,
    Seconds dt,
    Terrain terrain,
    List<AkerisEvent> into,
  ) {
    final tether = akeris._tether;
    if (!identical(tether, _following) || tether is! PulledToward) {
      return akeris._retether(terrain);
    }
    final tuning = akeris._tuning;
    final target = Vec2(tether.bite.x, tether.bite.y + tuning.hangBelowBite);
    _elapsed = _elapsed + dt;
    final u = _smooth(_elapsed.value / tuning.pullAcrossTime.value);
    final body = akeris._body;
    body.sweepBy(_from + (target - _from).scaled(u) - body.feet, terrain);
    return _elapsed < tuning.pullAcrossTime
        ? this
        : akeris._postures.wallHanging;
  }
}
```

```dart
// lib/features/akeris/domain/climbing.dart — new
part of 'akeris.dart';

// A climb stretch on the chain, 50 su a click, or the mantle over the top
// edge (hook-blades.md §3; animations.md §4 wall climb, mantle; the
// sandbox's `startWallStab`, `wallClimb`, `mantle`).
final class _Climbing extends _Posture {
  Tether _following = const Slack();
  Vec2 _from = Vec2.zero;
  Vec2? _target;
  bool _isMantle = false;
  Seconds _elapsed = Seconds.zero;

  @override
  Stance stanceOf(Akeris akeris) => Stance.climbing;

  @override
  bool get isTethered => true;

  @override
  AkerisEvent? enter(Akeris akeris) {
    _following = akeris._tether;
    _from = akeris._body.feet;
    _target = null;
    _elapsed = Seconds.zero;
    akeris._body
      ..velocity = Vec2.zero
      ..isGrounded = false;
    return null;
  }

  @override
  _Posture step(
    Akeris akeris,
    Seconds dt,
    Terrain terrain,
    List<AkerisEvent> into,
  ) {
    final tether = akeris._tether;
    if (!identical(tether, _following) || tether is! ClimbingTo) {
      final next = akeris._retether(terrain);
      if (identical(next, this)) enter(akeris);
      return next;
    }
    final target = _target ??= _targetFor(akeris, tether.bite, terrain);
    final time = _isMantle
        ? akeris._tuning.mantleTime
        : akeris._tuning.climbStretchTime;
    _elapsed = _elapsed + dt;
    final u = _smooth(_elapsed.value / time.value);
    final body = akeris._body;
    body.sweepBy(_from + (target - _from).scaled(u) - body.feet, terrain);
    if (_elapsed < time) return this;
    return _isMantle
        ? akeris._mantled(terrain, into)
        : akeris._postures.wallHanging;
  }

  Vec2 _targetFor(Akeris akeris, Vec2 bite, Terrain terrain) {
    final tuning = akeris._tuning;
    akeris._body.faceToward(bite.x);
    final side = akeris._body.facing.sign.toDouble();
    final probe = tuning.edgeProbe;
    _isMantle =
        terrain.isStaticSolidAt(bite + Vec2(side * probe, probe)) &&
        !terrain.isStaticSolidAt(bite + Vec2(side * probe, -probe));
    return _isMantle
        ? Vec2(bite.x + side * tuning.mantleOver, bite.y)
        : Vec2(bite.x, bite.y + tuning.stretchBelowBite);
  }
}
```

```dart
// lib/features/akeris/domain/wall_hanging.dart — new
part of 'akeris.dart';

// Hanging from a wall by his bitten blade, still until he climbs or jumps
// off (animations.md §4 wall hang; the sandbox's `wallHang`).
final class _WallHanging extends _Posture {
  Tether _following = const Slack();

  @override
  Stance stanceOf(Akeris akeris) => Stance.wallHanging;

  @override
  bool get isTethered => true;

  @override
  AkerisEvent? enter(Akeris akeris) {
    _following = akeris._tether;
    final body = akeris._body
      ..velocity = Vec2.zero
      ..isGrounded = false;
    if (_following case Hung(:final anchor)) {
      body.faceToward(anchor.x);
    }
    return null;
  }

  @override
  _Posture step(
    Akeris akeris,
    Seconds dt,
    Terrain terrain,
    List<AkerisEvent> into,
  ) {
    if (!identical(akeris._tether, _following)) {
      return akeris._retether(terrain);
    }
    if (!akeris._controls.takeJump()) {
      akeris._body.velocity = Vec2.zero;
      return this;
    }
    final tuning = akeris._tuning;
    final body = akeris._body..facing = akeris._body.facing.opposite;
    body.velocity = Vec2(
      body.facing.sign * tuning.wallJumpAway,
      -tuning.wallJumpLift,
    );
    return akeris._release(into);
  }
}
```
