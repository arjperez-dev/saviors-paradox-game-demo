# T120 — Roll Akeris on the ground and in the air

## Description

The roll and the air roll (VISION.md §Movement, `controls.md` G18).
- **Shift on the ground** rolls him 0.46 s, from 980 to 440 su/s, at 112 su tall and invulnerable
  the first 0.32 s. Space while rolling jumps out, keeping the roll's speed across. At the end he is
  on foot, or crouched under a low ceiling or with Left Ctrl held.
- **Shift in the air** is the air roll, once per jump, 0.42 s, the same invulnerability, keeping the
  jump's path; landing gives it back.

Both raise `RollStarted`. The application hears it to bring the chains back, since a roll cuts a
chain action (VISION.md).

**Invariants added:** rolls are at least 0.12 s apart; at most one air roll between two landings.
**Domain event:** `RollStarted(at)`.

```dart
// lib/features/akeris/domain/rolling.dart — new
part of 'akeris.dart';

// The roll: 0.46 s, invulnerable its first 0.32 s, cut short by a jump
// (VISION.md §Movement; animations.md §4 roll; the sandbox's roll branch).
final class _Rolling extends _Posture {
  Seconds _elapsed = Seconds.zero;
  Seconds _invulnerableFor = Seconds.zero;

  @override
  Stance stanceOf(Akeris akeris) => Stance.rolling;

  @override
  bool get isInvulnerable => _elapsed < _invulnerableFor;

  @override
  AkerisEvent? enter(Akeris akeris) {
    final tuning = akeris._tuning;
    _elapsed = Seconds.zero;
    _invulnerableFor = tuning.rollInvulnerability;
    akeris
      .._rollCooldown = tuning.rollCooldown
      .._body.height = tuning.rollingHeight
      .._body.face(akeris._controls.heading);
    return RollStarted(at: akeris._body.feet);
  }

  @override
  _Posture step(
    Akeris akeris,
    Seconds dt,
    Terrain terrain,
    List<AkerisEvent> into,
  ) {
    final tuning = akeris._tuning;
    final body = akeris._body;
    final u = (_elapsed.value / tuning.rollTime.value).clamp(0.0, 1.0);
    if (akeris._controls.takeJump() &&
        body.hasRoomFor(tuning.standingHeight, terrain)) {
      return _jumpOut(akeris, u, into);
    }
    body
      ..velocity = Vec2(
        body.facing.sign *
            _lerp(tuning.rollStartSpeed, tuning.rollEndSpeed, u),
        body.velocity.y,
      )
      ..fall(dt)
      ..move(dt, terrain);
    _elapsed = _elapsed + dt;
    if (!body.isGrounded) {
      body.height = tuning.standingHeight;
      return akeris._postures.airborne.walkOff(akeris);
    }
    return _elapsed >= tuning.rollTime ? _end(akeris, terrain) : this;
  }

  _Posture _jumpOut(Akeris akeris, double u, List<AkerisEvent> into) {
    final tuning = akeris._tuning;
    akeris._body.height = tuning.standingHeight;
    final kept = math.max(
      tuning.airSpeed,
      _lerp(tuning.rollJumpStartSpeed, tuning.rollJumpEndSpeed, u),
    );
    return akeris._postures.airborne.launch(akeris, into, keptSpeed: kept);
  }

  _Posture _end(Akeris akeris, Terrain terrain) {
    final standing = akeris._tuning.standingHeight;
    if (akeris._controls.isCrouchHeld ||
        !akeris._body.hasRoomFor(standing, terrain)) {
      return akeris._postures.crouching;
    }
    akeris._body.height = standing;
    return akeris._postures.onFoot;
  }
}
```

```dart
// lib/features/akeris/domain/air_rolling.dart — new
part of 'akeris.dart';

// The air roll: once per jump, keeping the jump's path, with the roll's
// invulnerability (controls.md G18; animations.md §4 air roll).
final class _AirRolling extends _Posture {
  Seconds _elapsed = Seconds.zero;
  Seconds _invulnerableFor = Seconds.zero;

  @override
  Stance stanceOf(Akeris akeris) => Stance.airRolling;

  @override
  bool get isInvulnerable => _elapsed < _invulnerableFor;

  @override
  AkerisEvent? enter(Akeris akeris) {
    _elapsed = Seconds.zero;
    _invulnerableFor = akeris._tuning.rollInvulnerability;
    akeris
      .._hasAirRolled = true
      .._body.face(akeris._controls.heading);
    return RollStarted(at: akeris._body.feet);
  }

  @override
  _Posture step(
    Akeris akeris,
    Seconds dt,
    Terrain terrain,
    List<AkerisEvent> into,
  ) {
    final body = akeris._body..fall(dt);
    final fallSpeed = body.velocity.y;
    body.move(dt, terrain);
    _elapsed = _elapsed + dt;
    if (body.isGrounded) {
      return akeris._postures.airborne.land(akeris, fallSpeed, into);
    }
    return _elapsed >= akeris._tuning.airRollTime
        ? akeris._postures.airborne
        : this;
  }
}
```
