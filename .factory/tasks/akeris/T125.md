# T125 — Swing Akeris from a ring on his chain

## Description

Hanging from a ring (`hook-blades.md` §3). When the tether becomes `Hung(hold: ring)`, Akeris swings
as a pendulum about the ring from his gripping hand, 218.6 su above his feet. The chain reels in to
the tether's length at 420 su/s. A/D pump the swing; it is damped at 0.12/s and held within
1.45 rad. Space lets go, adding 380 su/s upward. Stone in his way drops him (the sandbox's `swing`).
A hit never makes him let go (VISION.md).

On foot and in the air he enters the tethered posture the tether names. On a `Slack` tether he
falls, or stands where he is grounded. Letting go raises `ChainReleased`, which the application
turns into `ChainBlades.letGo()`, and the root's tether is slack at once.

**Invariants added:**
- His hand stays on the circle of the chain's length about the ring.
- Letting go always leaves the tether slack.

**Domain event:** `ChainReleased()`.

## Constraints

- As T118. No `as`: the tether is read with a pattern (coding-standards §6).

```dart
// lib/features/akeris/domain/swinging.dart — new
part of 'akeris.dart';

// Hanging from a ring: a pendulum about it that A/D pump (hook-blades.md
// §3; animations.md §4 swing; the sandbox's `enterSwing` and `swing`).
final class _Swinging extends _Posture {
  Tether _following = const Slack();
  double _angle = 0;
  double _spin = 0;
  double _length = 0;

  @override
  Stance stanceOf(Akeris akeris) => Stance.swinging;

  @override
  bool get isTethered => true;

  @override
  AkerisEvent? enter(Akeris akeris) {
    final tether = akeris._tether;
    if (tether case Hung(:final anchor)) {
      final body = akeris._body;
      final hand = _handOf(akeris);
      final arm = hand - anchor;
      _following = tether;
      _length = math.max(akeris._tuning.shortestChain, arm.length);
      _angle = math.atan2(arm.x, arm.y);
      _spin =
          (body.velocity.x * math.cos(_angle) -
              body.velocity.y * math.sin(_angle)) /
          _length;
      body.isGrounded = false;
    }
    return null;
  }

  @override
  _Posture step(
    Akeris akeris,
    Seconds dt,
    Terrain terrain,
    List<AkerisEvent> into,
  ) {
    final tether = akeris._tether;
    if (!identical(tether, _following) || tether is! Hung) {
      return akeris._retether(terrain);
    }
    if (akeris._controls.takeJump()) {
      return _letGo(akeris, into, lift: akeris._tuning.letGoLift);
    }
    _pump(akeris, dt, tether.length);
    final to = tether.anchor +
        Vec2(_length * math.sin(_angle), _length * math.cos(_angle)) +
        Vec2(0, akeris._tuning.handAboveFeet);
    final body = akeris._body;
    if (terrain.travel(body.box, to - body.feet) < 1) {
      body.velocity = Vec2(body.velocity.x * 0.3, 0); // the sandbox's stop
      return _letGo(akeris, into, lift: 0);
    }
    body
      ..feet = to
      ..velocity = Vec2(
        _length * _spin * math.cos(_angle),
        -_length * _spin * math.sin(_angle),
      );
    return this;
  }

  void _pump(Akeris akeris, Seconds dt, double length) {
    final tuning = akeris._tuning;
    final t = dt.value;
    _spin +=
        (-tuning.gravity / _length * math.sin(_angle) +
            akeris._controls.axis * tuning.swingPump * math.cos(_angle)) *
        t;
    _spin *= math.exp(-t * tuning.swingDamping);
    _angle += _spin * t;
    if (_angle.abs() > tuning.swingLimit) {
      _angle = _angle.sign * tuning.swingLimit;
      _spin *= -tuning.swingBounce;
    }
    final reel = tuning.reelSpeed * t;
    _length += (length - _length).clamp(-reel, reel);
  }

  static Vec2 _handOf(Akeris akeris) =>
      akeris._body.feet - Vec2(0, akeris._tuning.handAboveFeet);

  _Posture _letGo(Akeris akeris, List<AkerisEvent> into, {required double lift}) {
    final body = akeris._body;
    body.velocity = Vec2(body.velocity.x, body.velocity.y - lift);
    akeris._tether = const Slack();
    into.add(const ChainReleased());
    return akeris._postures.airborne.fall();
  }
}
```

```dart
// posture.dart — `_Postures` gains `final _Swinging swinging = _Swinging();`
// akeris.dart — the part, and:

  // The tethered posture the tether names; null on a slack one.
  _Posture? _tethered() => switch (_tether) {
    Slack() => null,
    Hung(hold: Hold.ring) => _postures.swinging,
    Hung(hold: Hold.wall) || PulledToward() || ClimbingTo() => null,
  };

  /// What follows when the tether changed: its own posture, or on foot or
  /// in the air on a slack one.
  _Posture _retether(Terrain terrain) {
    final tethered = _tethered();
    if (tethered != null) {
      return tethered;
    }
    _body.feelGround(terrain);
    return _body.isGrounded ? _postures.onFoot : _postures.airborne.fall();
  }

// on_foot.dart and airborne.dart — first thing in `step`:
    final tethered = akeris._tethered();
    if (tethered != null) {
      return tethered;
    }
```
