# T089 — Model a room template as values

## Description

The values dungeon builds zones from:
- the grid (`layout.md` §B.1–§B.2: 22 × 8 cells of 1600 × 1024 su, the floor 700 su below a row's
  top, passages 260 su tall, walls 50 su wide as the sandbox's `ZC` draws them);
- a room template and its parts: passages, islands, walls, ring places, drift paths, spots, and
  the room kind;
- the zone's id blocks (operator, 2026-10-16: `1 000 000 × zone + n`, rooms from 0, rings from
  1000, altars from 2000, spirits from 3000, platforms from 4000; no contract hands ids out yet);
- `edgesBox`, the one home of turning four edges into a `WorldBox`.

Every part mirrors itself across its room's width, as `layout.md` §B.3 lets every template be
mirrored. A drift path gives its platform's box at any time: eased out and back, `(1 − cos)/2`
along its travel over its period, from its phase (`hazards.md`; the sandbox's `zDriftPos`; 44 su
thick, as the sandbox's `zoneBuild` makes it).

## Constraints

- Every type is a value object: immutable, `final` fields, a `const` constructor wherever it
  compiles (not where an assert reads a `Seconds`, nor where a list is made unmodifiable).
  - `==` is the same runtime type with every field equal, lists element-wise.
  - `hashCode` is `Object.hash` of the fields in declaration order, with `Object.hashAll` for a
    list.
  - Lists leave a template unmodifiable.
- `RoomKind` and `SpotKind` are plain enums with no tests of their own.
- A test asserting a reference's value names the reference in a comment at its `expect`.

/// The grid a zone is laid on (layout.md §B.1, §B.2; the sandbox's `ZC`).

  const ZoneGrid({
    required this.columns,
    required this.rows,
    required this.cellWidth,
    required this.cellHeight,
    required this.floorDepth,
    required this.passageHeight,
    required this.wallWidth,

       assert(cellWidth > 0 && cellHeight > 0, 'a cell has a size');

  /// 700 su below a row's top, passages 260 su tall; `ZC.wall`: 50.

    rows: 8,
    cellWidth: 1600,
    cellHeight: 1024,

  final int rows;
  final double cellWidth;
  final double cellHeight;
  final double floorDepth;
  final double passageHeight;
  final double wallWidth;

      other.rows == rows &&
      other.cellWidth == cellWidth &&
      other.cellHeight == cellHeight &&
      other.floorDepth == floorDepth &&
      other.passageHeight == passageHeight &&
      other.wallWidth == wallWidth;

  @override
  int get hashCode => Object.hash(
    columns,
    rows,
    cellWidth,
    cellHeight,
    floorDepth,
    passageHeight,
    wallWidth,
  );
}

  const ZoneIds(this.zone);

  final ZoneNumber zone;

  EntityId room(int index) => _at(0, index);

  EntityId ring(int index) => _at(1000, index);

  EntityId altar(int index) => _at(2000, index);

  EntityId spirit(int index) => _at(3000, index);

  EntityId platform(int index) => _at(4000, index);

    assert(index >= 0 && index < 1000, 'an index fits its block');
    return EntityId(zone.value * 1000000 + block + index);
  }

  @override
  bool operator ==(Object other) => other is ZoneIds && other.zone == zone;

  @override
  int get hashCode => zone.hashCode;
}

// lib/features/dungeon/domain/drift_path.dart — new
import 'dart:math' as math;

import 'package:saviors_paradox/core/contracts/kernel/seconds.dart';
import 'package:saviors_paradox/core/contracts/kernel/vec2.dart';
import 'package:saviors_paradox/core/contracts/kernel/world_box.dart';
import 'package:saviors_paradox/features/dungeon/domain/edges_box.dart';

/// The path of a drifting platform, in su from its room's top left.
final class DriftPath {
  DriftPath({
    required this.left,
    required this.top,
    required this.width,
    required this.travel,
    required this.period,
  }) : assert(width > 0, 'a platform has a width'),
       assert(period.value > 0, 'a platform drifts over some time');

  /// The sandbox's `zoneBuild`: a drifting platform is 44 su thick.
  static const double thickness = 44;

  final double left;
  final double top;
  final double width;
  final Vec2 travel;
  final Seconds period;

  DriftPath mirrored(double roomWidth) => DriftPath(
    left: roomWidth - left - width,
    top: top,
    width: width,
    travel: Vec2(-travel.x, travel.y),
    period: period,
  );

  /// hazards.md: eased out and back along one line, `(1 − cos)/2`, never
  /// a jerk (the sandbox's `zDriftPos`).
  WorldBox boxAt({
    required Vec2 origin,
    required double phase,
    required Seconds time,
  }) {
    final turn = time.value / period.value + phase;
    final eased = (1 - math.cos(2 * math.pi * turn)) / 2;
    final x = origin.x + left + travel.x * eased;
    final y = origin.y + top + travel.y * eased;
    return edgesBox(x, y, x + width, y + thickness);
  }

  @override
  bool operator ==(Object other) =>
      other is DriftPath &&
      other.left == left &&
      other.top == top &&
      other.width == width &&
      other.travel == travel &&
      other.period == period;

  @override
  int get hashCode => Object.hash(left, top, width, travel, period);
}

// lib/features/dungeon/domain/room_template.dart — new
import 'package:saviors_paradox/core/contracts/kernel/vec2.dart';
import 'package:saviors_paradox/features/dungeon/domain/drift_path.dart';
import 'package:saviors_paradox/features/dungeon/domain/island_span.dart';
import 'package:saviors_paradox/features/dungeon/domain/passage.dart';
import 'package:saviors_paradox/features/dungeon/domain/room_kind.dart';
import 'package:saviors_paradox/features/dungeon/domain/room_spot.dart';
import 'package:saviors_paradox/features/dungeon/domain/spot_kind.dart';
import 'package:saviors_paradox/features/dungeon/domain/wall_span.dart';
import 'package:saviors_paradox/features/dungeon/domain/zone_grid.dart';

/// One hand-drawn room of a biome's catalog (layout.md §B.3).
final class RoomTemplate {
  RoomTemplate({
    required this.id,
    required this.kind,
    required this.cellsWide,
    required this.cellsHigh,
    required List<Passage> passages,
    required List<IslandSpan> islands,
    List<WallSpan> walls = const [],
    List<Vec2> rings = const [],
    List<DriftPath> drifts = const [],
    List<RoomSpot> spots = const [],
    this.hasChasm = false,
  }) : assert(cellsWide >= 1 && cellsHigh >= 1, 'a room fills a cell'),
       passages = List.unmodifiable(passages),
       islands = List.unmodifiable(islands),
       walls = List.unmodifiable(walls),
       rings = List.unmodifiable(rings),
       drifts = List.unmodifiable(drifts),
       spots = List.unmodifiable(spots);

  final String id;
  final RoomKind kind;
  final int cellsWide;
  final int cellsHigh;
  final List<Passage> passages;
  final List<IslandSpan> islands;
  final List<WallSpan> walls;
  final List<Vec2> rings;
  final List<DriftPath> drifts;
  final List<RoomSpot> spots;
  final bool hasChasm;

  double widthOn(ZoneGrid grid) => cellsWide * grid.cellWidth;

  double heightOn(ZoneGrid grid) => cellsHigh * grid.cellHeight;

  /// The same room drawn the other way round (layout.md §B.3).
  RoomTemplate mirrored(ZoneGrid grid) {
    final width = widthOn(grid);
    return RoomTemplate(
      id: id,
      kind: kind,
      cellsWide: cellsWide,
      cellsHigh: cellsHigh,
      passages: [for (final passage in passages) passage.mirrored()],
      islands: [for (final island in islands) island.mirrored(width)],
      walls: [for (final wall in walls) wall.mirrored(width)],
      rings: [for (final ring in rings) Vec2(width - ring.x, ring.y)],
      drifts: [for (final drift in drifts) drift.mirrored(width)],
      spots: [for (final spot in spots) spot.mirrored(width)],
      hasChasm: hasChasm,
    );
  }

  /// Where the spot of [kind] stands, or null when the room has none.
  Vec2? spot(SpotKind kind) {
    for (final spot in spots) {
      if (spot.kind == kind) {
        return spot.position;
      }
    }
    return null;
  }

  @override
  bool operator ==(Object other) =>
      other is RoomTemplate &&
      other.id == id &&
      other.kind == kind &&
      other.cellsWide == cellsWide &&
      other.cellsHigh == cellsHigh &&
      _same(other.passages, passages) &&
      _same(other.islands, islands) &&
      _same(other.walls, walls) &&
      _same(other.rings, rings) &&
      _same(other.drifts, drifts) &&
      _same(other.spots, spots) &&
      other.hasChasm == hasChasm;

  @override
  int get hashCode => Object.hash(
    id,
    kind,
    cellsWide,
    cellsHigh,
    Object.hashAll(passages),
    Object.hashAll(islands),
    Object.hashAll(walls),
    Object.hashAll(rings),
    Object.hashAll(drifts),
    Object.hashAll(spots),
    hasChasm,
  );
}

bool _same<T>(List<T> a, List<T> b) {
  if (a.length != b.length) {
    return false;
  }
  for (var i = 0; i < a.length; i++) {
    if (a[i] != b[i]) {
      return false;
    }
  }
  return true;
}
